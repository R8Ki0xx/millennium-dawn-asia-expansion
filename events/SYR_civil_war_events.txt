# 多方混战：叛军保留了自己的身份，60~90 天后 FSA 与 NUS 在地方上冒出来（作用域：SYR）
country_event = {
	id = SYR_terror.5
	title = SYR_terror.5.t
	desc = SYR_terror.5.d
	picture = GFX_SYR_ssnp
	is_triggered_only = yes

	trigger = {
		has_global_flag = GLOBAL_syrian_civil_war_started
		NOT = { has_global_flag = GLOBAL_syrian_civil_war_over }
	}

	option = {
		name = SYR_terror.5.a
		log = "[GetDateText]: [This.GetName]: SYR_terror.5.a executed"
		ai_chance = { base = 100 }
		hidden_effect = {
			log = "[GetDateText]: SYR_civil_war: SYR_terror.5 releases FSA and NUS"
			SYR_cw_release_fsa = yes
			SYR_cw_release_nus = yes
			SYR_cw_declare_wars = yes
			SYR_terror_publish_all = yes
		}
		custom_effect_tooltip = SYR_terror_tt_factions
	}
}

	}

		trigger = { has_global_flag = SYR_cw_path_moderate }
	}
	desc = {
		text = SYR_terror.6.d_fsa
		trigger = { has_global_flag = SYR_cw_path_salafi }

	picture = GFX_SYR_tahrir_al_sham
	is_triggered_only = yes

	trigger = {
		has_global_flag = GLOBAL_syrian_civil_war_started
		NOT = { has_global_flag = GLOBAL_syrian_civil_war_over }
	}

	option = {
		name = SYR_terror.6.a
		log = "[GetDateText]: [This.GetName]: SYR_terror.6.a executed"

			}
			SYR_cw_declare_wars = yes
			SYR_terror_publish_all = yes
		}
		custom_effect_tooltip = SYR_terror_tt_factions
	}
}
