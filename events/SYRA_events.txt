add_namespace = SYRA

# 阿萨德战后线：“开放大选”之后的剧情事件。
# 流程：国策 SYRA_open_elections → SYRA.1（大选前夜，触发千禧的竞选流程）→ 千禧选举事件 → on_daily_SYR 发现选举结束
#       → SYRA_election_result（common/scripted_effects/SYRA_effects.txt）：执政党没换 → SYRA.3；换了 → SYRA.4（复兴党是否下台）
#       → 国策 SYRA_baath_steps_down → SYRA.5；各执政党的后续国策 → SYRA.6。
# 旗标：SYRA_elections_open / SYRA_campaign_on / SYRA_election_done；结果 SYRA_res_baath / dem / neu / nat / fas / left；
#       下台方式 SYRA_handover_peaceful / SYRA_handover_bargain / SYRA_handover_refused。变量 SYRA_party_before = 开选举时的执政党编号。

# 1. 大选前夜
country_event = {
	id = SYRA.1
	title = SYRA.1.t
	desc = SYRA.1.d
	picture = GFX_election_day
	is_triggered_only = yes

	immediate = { set_country_flag = SYRA_campaign_on }

	# 公平竞选
	option = {
		name = SYRA.1.a
		log = "[GetDateText]: [Root.GetName]: SYRA.1.a executed"
		ai_chance = { base = 55 }
		add_stability = 0.02
		add_political_power = -25
		hidden_effect = { country_event = { id = MD4_election_campaign.0 days = 1 } }
	}
	# 保住复兴党的优势
	option = {
		name = SYRA.1.b
		log = "[GetDateText]: [Root.GetName]: SYRA.1.b executed"
		ai_chance = { base = 45 }
		add_political_power = 50
		add_stability = -0.03
		set_temp_variable = { party_index = SYRA_party_before }
		set_temp_variable = { party_popularity_increase = 0.03 }
		set_temp_variable = { temp_outlook_increase = 0.03 }
		change_relative_party_popularity = yes
		hidden_effect = { country_event = { id = MD4_election_campaign.0 days = 1 } }
	}
}

# 4. 复兴党落选：交不交权
country_event = {
	id = SYRA.4
	title = SYRA.4.t
	desc = SYRA.4.d
	picture = GFX_election_day
	is_triggered_only = yes

	# 体面交接
	option = {
		name = SYRA.4.a
		log = "[GetDateText]: [Root.GetName]: SYRA.4.a executed"
		ai_chance = { base = 50 }
		set_country_flag = SYRA_handover_peaceful
		add_stability = 0.03
		add_political_power = 50
	}
	# 讨价还价：军队与情报系统保留否决权
	option = {
		name = SYRA.4.b
		log = "[GetDateText]: [Root.GetName]: SYRA.4.b executed"
		ai_chance = { base = 35 }
		set_country_flag = SYRA_handover_bargain
		add_political_power = -25
		if = { limit = { NOT = { has_idea = SYRA_old_guard_veto_idea } } add_ideas = SYRA_old_guard_veto_idea }
	}
	# 拒绝下台：宣布选举无效
	option = {
		name = SYRA.4.c
		log = "[GetDateText]: [Root.GetName]: SYRA.4.c executed"
		ai_chance = { base = 15 }
		set_country_flag = SYRA_handover_refused
		set_country_flag = Second_Syrian_Civil_War
		start_civil_war = {
			ideology = nationalist
			size = 0.4
		}
	}
}

# 5. 交接仪式（国策 SYRA_baath_steps_down）

	id = SYRA.5
	title = SYRA.5.t
	desc = {
		text = SYRA.5.d_bargain
		trigger = { has_country_flag = SYRA_handover_bargain }
	}
	desc = {
		text = SYRA.5.d_peaceful
		trigger = { NOT = { has_country_flag = SYRA_handover_bargain } }
	}
	picture = GFX_Assad_parade
	is_triggered_only = yes

	option = {
		name = SYRA.5.a
		log = "[GetDateText]: [Root.GetName]: SYRA.5.a executed"
		add_political_power = 25
		if = {
			limit = { has_country_flag = SYRA_handover_peaceful }
			add_stability = 0.02
		}
	}
}
