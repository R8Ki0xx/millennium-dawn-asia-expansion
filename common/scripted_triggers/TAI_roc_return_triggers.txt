	NOT = {

		state = 1156
		state = 1157
		state = 1158

	}
}

# Country scope (TAI): is the window for a military return to the mainland open?
TAI_return_window_open = {
	custom_trigger_tooltip = {
		tooltip = TAI_return_window_tt
		OR = {
			check_variable = { TAI_mainland_unrest > 49 }
			CHI = { has_stability < 0.35 }
			CHI = { has_civil_war = yes }
			any_country = {
				is_major = yes
				NOT = { tag = TAI }
				has_war_with = CHI
			}
		}
	}
}

# Country scope: countries covered by the East Asian labour-market correction（v3.19：只剩中国大陆、台湾与中共残余）
EA_labor_country = {
	any_owned_state = { always = yes }
	OR = {
		original_tag = CHI
		original_tag = TAI
		original_tag = PRC
	}
}
