# MD ROC Return v3.36 —— 选举系统的启动（由 _tools/v336/vel_events.py 生成）
on_actions = {
	on_new_term_election = {
		effect = {
			if = {
				limit = { original_tag = USA }
				country_event = { id = VEL_USA.1 }
			}
		}
	}
	# 开局就启用：否则要等游戏过完第一天，决议栏里的选举界面才有按钮和数据
	on_startup = {
		effect = {
			if = {
				limit = { country_exists = USA }
				USA = {
					if = {
						limit = { NOT = { has_country_flag = VEL_USA_on } }
						set_country_flag = VEL_USA_on
						set_country_flag = generic_election_killswitch
						VEL_USA_init = yes
					}
				}
			}
		}
	}
	on_daily_USA = {
		effect = {
			if = {
				limit = { NOT = { has_country_flag = VEL_USA_on } }
				set_country_flag = VEL_USA_on
				set_country_flag = generic_election_killswitch
				VEL_USA_init = yes
			}
			VEL_USA_poll_tick = yes
		}
	}
	on_daily_CHI = {
		effect = {
			# 光复之后的中华民国，换上可视化选举，并停掉千禧的通用选举
			if = {
				limit = { has_country_flag = ROC_new_chi NOT = { has_country_flag = VEL_CHI_on } }
				set_country_flag = VEL_CHI_on
				set_country_flag = generic_election_killswitch
			}
			if = {
				limit = { has_country_flag = VEL_CHI_on NOT = { has_country_flag = VEL_CHI_inited } }
				VEL_CHI_init = yes
			}
			if = {
				limit = { has_country_flag = VEL_CHI_on has_country_flag = VEL_CHI_inited }
				VEL_CHI_poll_tick = yes
			}
		}
	}
}
