# Scope: CHI。按进度重算新疆的州修正
TAI_fr_xj_recalc = {
	set_temp_variable = { fr_gap = 100 }
	subtract_from_temp_variable = { fr_gap = TAI_fr_xj }
	divide_temp_variable = { fr_gap = 100 }
	clamp_temp_variable = { var = fr_gap min = 0 max = 1 }
	set_variable = { TAI_fr_xj_garrison = fr_gap }
	multiply_variable = { TAI_fr_xj_garrison = 0.6 }
	set_variable = { TAI_fr_xj_res = fr_gap }
	multiply_variable = { TAI_fr_xj_res = 0.25 }
	set_variable = { TAI_fr_xj_comp = fr_gap }
	multiply_variable = { TAI_fr_xj_comp = -0.2 }
	set_variable = { TAI_fr_xj_fac = fr_gap }
	multiply_variable = { TAI_fr_xj_fac = -0.3 }
	every_owned_state = {
		limit = { OR = { state = 592 state = 591 } }
		if = {
			limit = { check_variable = { CHI.TAI_fr_xj < 100 } }
			if = { limit = { NOT = { has_dynamic_modifier = { modifier = TAI_frontier_xj_modifier } } } add_dynamic_modifier = { modifier = TAI_frontier_xj_modifier } }
		}
		else = {
			if = { limit = { has_dynamic_modifier = { modifier = TAI_frontier_xj_modifier } } remove_dynamic_modifier = { modifier = TAI_frontier_xj_modifier } }
		}
	}
}

TAI_fr_xj_add = {

		add_to_variable = { TAI_fr_xj = fr_d }

			limit = { check_variable = { TAI_fr_xj > 99.9 } }
			set_country_flag = TAI_fr_xj_done
			country_event = { id = roc_return.161 days = 1 }
		}
		TAI_fr_xj_recalc = yes
	}
}
