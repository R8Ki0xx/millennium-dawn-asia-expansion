# Scope: the ROC (CHI after the tag switch). Ends the CCP's aid to Pyongyang and the old CHI-NKO cooperation ideas.
TAI_end_ccp_nko_aid = {
	if = { limit = { has_idea = CHI_china_economic_assistant } remove_ideas = CHI_china_economic_assistant }
	if = { limit = { has_idea = CHI_NKO_base } remove_ideas = CHI_NKO_base }
	if = { limit = { has_idea = CHI_NKO_border_trade_idea } remove_ideas = CHI_NKO_border_trade_idea }
	if = { limit = { has_idea = CHI_NKO_chinese_investments_idea } remove_ideas = CHI_NKO_chinese_investments_idea }
	if = { limit = { has_idea = CHI_NKO_energy_supply_idea } remove_ideas = CHI_NKO_energy_supply_idea }
	if = { limit = { has_idea = CHI_NKO_telecom_idea } remove_ideas = CHI_NKO_telecom_idea }
	if = {
		limit = { country_exists = NKO }
		NKO = {
			if = { limit = { has_idea = NKO_china_economic_assistant } remove_ideas = NKO_china_economic_assistant }
			if = { limit = { has_idea = CHI_NKO_base } remove_ideas = CHI_NKO_base }
			if = { limit = { has_idea = CHI_NKO_border_trade_idea } remove_ideas = CHI_NKO_border_trade_idea }
			if = { limit = { has_idea = CHI_NKO_chinese_investments_idea } remove_ideas = CHI_NKO_chinese_investments_idea }
			if = { limit = { has_idea = CHI_NKO_energy_supply_idea } remove_ideas = CHI_NKO_energy_supply_idea }
			if = { limit = { has_idea = CHI_NKO_telecom_idea } remove_ideas = CHI_NKO_telecom_idea }
			if = { limit = { has_dynamic_modifier = { modifier = CHI_NKO_shared_modifier } } remove_dynamic_modifier = { modifier = CHI_NKO_shared_modifier } }
			remove_opinion_modifier = { target = ROOT modifier = mutual_trade_agreement }
		}
	}
	set_country_flag = TAI_nko_aid_ended
}

# Scope: the ROC. Pro-China coup in Pyongyang: Kim Jong-nam (or a KPA junta) takes power and NKO becomes a puppet.
TAI_pyongyang_coup_effect = {
	NKO = {
		if = { limit = { has_idea = NKO_china_economic_assistant } remove_ideas = NKO_china_economic_assistant }
		kill_country_leader = yes
		if = {
			limit = { date < 2017.2.13 }
			create_country_leader = {
				name = "Kim Jong-nam"
				picture = "Kim Jong-nam.dds"
				desc = KIM_JONG_NAM_LEADER_DESC
				ideology = communist_state
				traits = {
					emerging_communist_state
					Reformer
					pro_china
				}
			}
		}
		else = {
			create_country_leader = {
				name = TAI_nko_junta_leader
				picture = "gen_Ri_Yong_Gil.dds"
				ideology = communist_state
				traits = {
					emerging_communist_state
					pro_china
				}
			}
		}
		add_ideas = TAI_nko_transition_idea
		add_opinion_modifier = { target = ROOT modifier = TAI_nko_detente }
	}
	puppet = NKO
}
