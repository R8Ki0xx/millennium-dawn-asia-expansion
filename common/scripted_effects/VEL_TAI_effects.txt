	set_variable = { VEL_TAI_sal^vel_mi = 2 }
	set_variable = { VEL_TAI_main = vel_issue }
	VEL_TAI_run_poll = yes
}

VEL_TAI_reset = {
	set_country_flag = VEL_TAI_open
	for_loop_effect = { start = 0 end = 22 value = vr set_variable = { VEL_TAI_rev^vr = 0 } set_variable = { VEL_TAI_camp^vr = 0 } }
	for_loop_effect = { start = 0 end = 6 value = vi set_variable = { VEL_TAI_sal^vi = 1 } }
	for_loop_effect = { start = 0 end = 7 value = vp set_variable = { VEL_TAI_cand^vp = 0 } set_variable = { VEL_TAI_run_seat^vp = 0 } set_variable = { VEL_TAI_run_nat^vp = 0 } }
	set_variable = { VEL_TAI_run_w = 0 }
	set_variable = { VEL_TAI_main = 0 }
	set_variable = { VEL_TAI_poll_ready = 0 }
	set_variable = { VEL_TAI_done = 0 }
	set_variable = { VEL_TAI_anim_n = 0 }
	set_variable = { VEL_TAI_view = 1 }
	set_variable = { VEL_TAI_phase = 1 }
	VEL_TAI_run_poll = yes
}

# 每天由 on_action 调用；每 30 天重算一次民调（开票进行中不刷新）
VEL_TAI_poll_tick = {
	if = { limit = { has_country_flag = VEL_TAI_inited } add_to_variable = { VEL_TAI_news_t = 1 } if = { limit = { check_variable = { VEL_TAI_news_t > 3 } } set_variable = { VEL_TAI_news_t = 0 } add_to_variable = { VEL_TAI_news_i = 1 } if = { limit = { check_variable = { VEL_TAI_news_i > 3 } } set_variable = { VEL_TAI_news_i = 0 } } } }
	if = {
		limit = { has_country_flag = VEL_TAI_inited check_variable = { global.num_days > VEL_TAI_poll_due } NOT = { check_variable = { VEL_TAI_phase = 2 } } }
		set_variable = { VEL_TAI_poll_due = global.num_days }
		add_to_variable = { VEL_TAI_poll_due = 30 }
		VEL_TAI_run_poll = yes
	}
}
