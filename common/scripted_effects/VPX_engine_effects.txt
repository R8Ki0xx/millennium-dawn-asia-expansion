# 最大余数法：输入 vpx_f^0..n-1（理论值）、temp vlr_n（项数）、vlr_t（总数）；输出 vpx_o^*
vpx_lr = {
	set_temp_variable = { vlr_sum = 0 }
	for_loop_effect = { start = 0 end = vlr_n value = vli
		set_temp_variable = { vlr_x = vpx_f^vli }
		set_temp_variable = { vlr_fl = vlr_x }
		round_temp_variable = vlr_fl
		if = { limit = { check_variable = { vlr_fl > vlr_x } } subtract_from_temp_variable = { vlr_fl = 1 } }
		if = { limit = { check_variable = { vlr_fl < 0 } } set_temp_variable = { vlr_fl = 0 } }
		set_variable = { vpx_o^vli = vlr_fl }
		set_variable = { vpx_rem^vli = vlr_x }
		subtract_from_variable = { vpx_rem^vli = vlr_fl }
		if = { limit = { check_variable = { vlr_x < 0.001 } } set_variable = { vpx_rem^vli = -1 } }
		add_to_temp_variable = { vlr_sum = vlr_fl }
	}
	set_temp_variable = { vlr_left = vlr_t }
	subtract_from_temp_variable = { vlr_left = vlr_sum }
	for_loop_effect = { start = 0 end = vlr_n value = vlk
		if = {
			limit = { check_variable = { vlr_left > 0.5 } }
			set_temp_variable = { vlr_best = -0.5 } set_temp_variable = { vlr_bj = -1 }
			for_loop_effect = { start = 0 end = vlr_n value = vli if = { limit = { check_variable = { vpx_rem^vli > vlr_best } } set_temp_variable = { vlr_best = vpx_rem^vli } set_temp_variable = { vlr_bj = vli } } }
			if = { limit = { check_variable = { vlr_bj > -0.5 } } add_to_variable = { vpx_o^vlr_bj = 1 } set_variable = { vpx_rem^vlr_bj = -1 } subtract_from_temp_variable = { vlr_left = 1 } }
		}
	}
	# 舍入误差剩下的席位给最大的一项
	if = {
		limit = { check_variable = { vlr_left > 0.5 } }
		set_temp_variable = { vlr_best = 0 } set_temp_variable = { vlr_bj = -1 }
		for_loop_effect = { start = 0 end = vlr_n value = vli if = { limit = { check_variable = { vpx_o^vli > vlr_best } } set_temp_variable = { vlr_best = vpx_o^vli } set_temp_variable = { vlr_bj = vli } } }
		if = { limit = { check_variable = { vlr_bj > -0.5 } } add_to_variable = { vpx_o^vlr_bj = vlr_left } }
	}
}
