# Scope: 即将被吞并的国家（旧 CHI 或 TAI）。记下它在各国的影响力份额（各国变量 TAI_roc_infl_rec，两次累加）
TAI_roc_v319_record_influence = {
	set_temp_variable = { roc_src = THIS.id }
	every_country = {
		limit = { NOT = { tag = CHI } NOT = { tag = TAI } is_in_array = { influence_array = roc_src } }
		for_each_loop = {
			array = influence_array
			value = v
			index = i
			if = {
				limit = { check_variable = { v = roc_src } }
				add_to_variable = { TAI_roc_infl_rec = influence_array_val^i }
			}
		}
	}
}

# Scope: 新 CHI。把记下的影响力写回各国的 CHI 条目
TAI_roc_v319_restore_influence = {
	set_temp_variable = { roc_chi = THIS.id }
	every_country = {
		limit = { NOT = { tag = CHI } has_variable = TAI_roc_infl_rec }
		set_temp_variable = { roc_add = TAI_roc_infl_rec }
		clamp_temp_variable = { var = roc_add min = 0 max = 80 }
		set_temp_variable = { roc_dst = -1 }
		for_each_loop = {
			array = influence_array
			value = v
			index = i
			if = { limit = { check_variable = { v = roc_chi } } set_temp_variable = { roc_dst = i } }
		}
		if = {
			limit = { check_variable = { roc_dst > -1 } }
			if = { limit = { check_variable = { influence_array_val^roc_dst < roc_add } } set_variable = { influence_array_val^roc_dst = roc_add } }
		}
		else_if = {
			limit = { check_variable = { roc_add > 0.01 } }
			add_to_array = { influence_array = roc_chi }
			add_to_array = { influence_array_val = roc_add }
		}
		recalculate_influence = yes
		clear_variable = TAI_roc_infl_rec
	}
}
