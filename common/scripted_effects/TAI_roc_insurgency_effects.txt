# Scope: ROC-CHI, monthly
TAI_insurgency_monthly = {
	TAI_insurgency_count = yes
	set_temp_variable = { ins_change = TAI_ins_severe }
	multiply_temp_variable = { ins_change = 0.5 }
	set_temp_variable = { ins_a = TAI_ins_active }
	multiply_temp_variable = { ins_a = 0.25 }
	add_to_temp_variable = { ins_change = ins_a }
	if = { limit = { country_exists = PRC } add_to_temp_variable = { ins_change = 2 } }
	if = { limit = { has_war_with = PRC } add_to_temp_variable = { ins_change = 2 } }
	if = { limit = { has_dynamic_modifier = { modifier = TAI_dbf_hearts } } add_to_temp_variable = { ins_change = TAI_dbf_hearts_lvl } }
	if = { limit = { has_stability > 0.6 } add_to_temp_variable = { ins_change = -2 } }
	if = { limit = { check_variable = { TAI_ins_severe < 1 } } add_to_temp_variable = { ins_change = -1 } }
	TAI_insurgency_change = yes
	# escalation / slow recovery
	if = {
		limit = { check_variable = { TAI_ccp_insurgency > 50 } }
		every_owned_state = { limit = { OR = { has_dynamic_modifier = { modifier = TAI_red_insurgency_2 } has_dynamic_modifier = { modifier = TAI_red_insurgency_1 } } } random = { chance = 5 TAI_insurgency_upgrade = yes } }
	}
	else_if = {
		limit = { check_variable = { TAI_ccp_insurgency < 20 } }
		every_owned_state = { limit = { OR = { has_dynamic_modifier = { modifier = TAI_red_insurgency_3 } has_dynamic_modifier = { modifier = TAI_red_insurgency_2 } has_dynamic_modifier = { modifier = TAI_red_insurgency_1 } } } random = { chance = 5 TAI_insurgency_downgrade = yes } }
	}
	TAI_insurgency_count = yes
	# the ROC crushed a PRC rising
	if = {
		limit = { has_country_flag = TAI_third_civil_war NOT = { country_exists = PRC } }
		clr_country_flag = TAI_third_civil_war
		set_temp_variable = { ins_change = -20 }
		TAI_insurgency_change = yes
		country_event = { id = roc_return.88 days = 1 }
	}
}
