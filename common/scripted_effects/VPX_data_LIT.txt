	clear_array = vpx_dot resize_array = { array = vpx_dot value = 1 size = 141 }
	vpx_setup = yes
}

vpx_enable_LIT = {
	if = {
		limit = { NOT = { has_country_flag = vpx_on } }
		set_country_flag = vpx_on
		if = { limit = { NOT = { has_country_flag = vpx_inited } } vpx_init_LIT = yes }
		else = { vpx_run_poll = yes }
	}
}
