	set_variable = { VELP_a_gov = 0 }
	for_loop_effect = { start = 0 end = 7 value = vi if = { limit = { check_variable = { VELP_a_g^vi = 1 } } add_to_variable = { VELP_a_gov = VELP_a_n^vi } } }
	set_variable = { VELP_a_need = VELP_a_tot } add_to_variable = { VELP_a_need = 1 } divide_variable = { VELP_a_need = 2 } add_to_variable = { VELP_a_need = 0.01 } round_variable = VELP_a_need
	set_temp_variable = { vcum = 0 } set_temp_variable = { vpos = 0 }
	for_loop_effect = { start = 0 end = 2 value = vside
		set_temp_variable = { vwant = 1 } subtract_from_temp_variable = { vwant = vside }
		for_loop_effect = { start = 0 end = 7 value = vi
			if = {
				limit = { check_variable = { VELP_a_g^vi = vwant } check_variable = { VELP_a_n^vi > 0 } check_variable = { VELP_a_tot > 0 } }
				add_to_temp_variable = { vcum = VELP_a_n^vi }
				set_temp_variable = { vend = vcum } multiply_temp_variable = { vend = 150 } divide_temp_variable = { vend = VELP_a_tot } round_temp_variable = vend
				if = { limit = { check_variable = { vend > 150 } } set_temp_variable = { vend = 150 } }
				set_temp_variable = { vfr = VELP_a_c^vi }
				if = { limit = { check_variable = { vwant = 1 } } add_to_temp_variable = { vfr = 1 } }
				else = { add_to_temp_variable = { vfr = 11 } }
				for_loop_effect = { start = vpos end = vend value = vk set_variable = { VELP_a_dot^vk = vfr } }
				if = { limit = { check_variable = { vend > vpos } } set_temp_variable = { vpos = vend } }
			}
		}
	}
	for_loop_effect = { start = vpos end = 150 value = vk set_variable = { VELP_a_dot^vk = 23 } }
}
