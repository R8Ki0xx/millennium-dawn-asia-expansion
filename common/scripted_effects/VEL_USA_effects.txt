# 输入：temp vel_noise（1 = 开票带随机，0 = 民调）；temp vel_gov = 执政党在本系统里的序号(0..P-1)。输出：res / win / best / sr / nat / seat 数组。
VEL_USA_compute = {
	set_temp_variable = { vel_gov = VEL_USA_gov }
	set_temp_variable = { vel_salsum = 0 }
	for_loop_effect = { start = 0 end = 6 value = vi add_to_temp_variable = { vel_salsum = VEL_USA_sal^vi } }
	for_loop_effect = {
		start = 0 end = 3 value = vp
		set_temp_variable = { vel_pi = VEL_USA_pidx^vp }
		set_temp_variable = { vel_f = party_pop_array^vel_pi }
		multiply_temp_variable = { vel_f = 3 }
		add_to_temp_variable = { vel_f = 0.05 }
		set_variable = { VEL_USA_f^vp = vel_f }
	}
	for_loop_effect = {
		start = 0 end = 8 value = vg
		set_temp_variable = { vel_sum = 0 }
		for_loop_effect = {
			start = 0 end = 3 value = vp
			set_temp_variable = { vel_sc = 0 }
			for_loop_effect = {
				start = 0 end = 6 value = vi
				set_temp_variable = { vel_ip = vp } multiply_temp_variable = { vel_ip = 6 } add_to_temp_variable = { vel_ip = vi }
				set_temp_variable = { vel_io = vg } multiply_temp_variable = { vel_io = 6 } add_to_temp_variable = { vel_io = vi }
				set_temp_variable = { vel_d = VEL_USA_pos^vel_ip }
				subtract_from_temp_variable = { vel_d = VEL_USA_op^vel_io }
				if = { limit = { check_variable = { vel_d < 0 } } multiply_temp_variable = { vel_d = -1 } }
				set_temp_variable = { vel_t = 10 }
				subtract_from_temp_variable = { vel_t = vel_d }
				multiply_temp_variable = { vel_t = VEL_USA_sal^vi }
				add_to_temp_variable = { vel_sc = vel_t }
			}
			divide_temp_variable = { vel_sc = vel_salsum }
			set_temp_variable = { vel_ia = vg } multiply_temp_variable = { vel_ia = 3 } add_to_temp_variable = { vel_ia = vp }
			add_to_temp_variable = { vel_sc = VEL_USA_aff^vel_ia }
			multiply_temp_variable = { vel_sc = 0.05 }
			add_to_temp_variable = { vel_sc = VEL_USA_cand^vp }
			if = { limit = { check_variable = { vel_sc < 0.02 } } set_temp_variable = { vel_sc = 0.02 } }
			multiply_temp_variable = { vel_sc = VEL_USA_f^vp }
			add_to_temp_variable = { vel_sum = vel_sc }
			set_variable = { VEL_USA_N^vel_ia = vel_sc }
		}
		for_loop_effect = {
			start = 0 end = 3 value = vp
			set_temp_variable = { vel_ia = vg } multiply_temp_variable = { vel_ia = 3 } add_to_temp_variable = { vel_ia = vp }
			divide_variable = { VEL_USA_N^vel_ia = vel_sum }
		}
	}
	for_loop_effect = { start = 0 end = 3 value = vp set_variable = { VEL_USA_nat^vp = 0 } set_variable = { VEL_USA_seat^vp = 0 } }
	for_loop_effect = {
		start = 0 end = 51 value = vr
		for_loop_effect = {
			start = 0 end = 3 value = vp
			set_temp_variable = { vel_ir = vr } multiply_temp_variable = { vel_ir = 3 } add_to_temp_variable = { vel_ir = vp }
			set_variable = { VEL_USA_res^vel_ir = 0 }
		}
		for_loop_effect = {
			start = 0 end = 8 value = vg
			set_temp_variable = { vel_igs = vr } multiply_temp_variable = { vel_igs = 8 } add_to_temp_variable = { vel_igs = vg }
			for_loop_effect = {
				start = 0 end = 3 value = vp
				set_temp_variable = { vel_ir = vr } multiply_temp_variable = { vel_ir = 3 } add_to_temp_variable = { vel_ir = vp }
				set_temp_variable = { vel_ia = vg } multiply_temp_variable = { vel_ia = 3 } add_to_temp_variable = { vel_ia = vp }
				set_temp_variable = { vel_x = VEL_USA_sh^vel_igs }
				multiply_temp_variable = { vel_x = VEL_USA_N^vel_ia }
				add_to_variable = { VEL_USA_res^vel_ir = vel_x }
			}
		}
		set_temp_variable = { vel_tot = 0 }
		for_loop_effect = {
			start = 0 end = 3 value = vp
			set_temp_variable = { vel_ir = vr } multiply_temp_variable = { vel_ir = 3 } add_to_temp_variable = { vel_ir = vp }
			set_temp_variable = { vel_v = 1 }
			add_to_temp_variable = { vel_v = VEL_USA_lean^vel_ir }
			multiply_temp_variable = { vel_v = VEL_USA_res^vel_ir }
			if = {
				limit = { check_variable = { vp = vel_gov } }
				set_temp_variable = { vel_c = VEL_USA_camp^vr }
				multiply_temp_variable = { vel_c = 0.04 }
				add_to_temp_variable = { vel_c = 1 }
				multiply_temp_variable = { vel_v = vel_c }
			}
			if = {
				limit = { check_variable = { vel_noise = 1 } }
				set_temp_variable_to_random = { var = vel_nr min = 0 max = 1 }
				multiply_temp_variable = { vel_nr = 0.06 }
				add_to_temp_variable = { vel_nr = 0.97 }
				multiply_temp_variable = { vel_v = vel_nr }
			}
			set_variable = { VEL_USA_res^vel_ir = vel_v }
			add_to_temp_variable = { vel_tot = vel_v }
		}
		set_temp_variable = { vel_best = 0 } set_temp_variable = { vel_bi = 0 }
		for_loop_effect = {
			start = 0 end = 3 value = vp
			set_temp_variable = { vel_ir = vr } multiply_temp_variable = { vel_ir = 3 } add_to_temp_variable = { vel_ir = vp }
			divide_variable = { VEL_USA_res^vel_ir = vel_tot }
			set_temp_variable = { vel_cur = VEL_USA_res^vel_ir }
			if = { limit = { check_variable = { vel_cur > vel_best } } set_temp_variable = { vel_best = vel_cur } set_temp_variable = { vel_bi = vp } }
		}
		set_variable = { VEL_USA_win^vr = vel_bi }
		add_to_variable = { VEL_USA_win^vr = 1 }
		set_variable = { VEL_USA_best^vr = vel_best }
		set_temp_variable = { vel_sumseat = 0 }
		for_loop_effect = {
			start = 0 end = 3 value = vp
			set_temp_variable = { vel_ir = vr } multiply_temp_variable = { vel_ir = 3 } add_to_temp_variable = { vel_ir = vp }
			set_temp_variable = { vel_x = VEL_USA_w^vr }
			multiply_temp_variable = { vel_x = VEL_USA_res^vel_ir }
			add_to_variable = { VEL_USA_nat^vp = vel_x }
			set_variable = { VEL_USA_sr^vel_ir = 0 }
		}
		set_temp_variable = { vel_wi = VEL_USA_win^vr }
		subtract_from_temp_variable = { vel_wi = 1 }
		set_temp_variable = { vel_ir = vr } multiply_temp_variable = { vel_ir = 3 } add_to_temp_variable = { vel_ir = vel_wi }
		set_variable = { VEL_USA_sr^vel_ir = VEL_USA_seatsr^vr }
		add_to_variable = { VEL_USA_seat^vel_wi = VEL_USA_seatsr^vr }
	}
}
