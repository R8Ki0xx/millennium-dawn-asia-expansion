# Scope: ROC-CHI。立委选举：席次 = 225 × 政党支持度
TAI_ly_election = {
	set_variable = { TAI_ly_kmt = party_pop_array^1 }

# Scope: ROC-CHI，每月：立委改选、执政党更替后提名新阁揆、少数政府的倒阁风险
TAI_gov_monthly = {
	TAI_gov_president_check = yes
	if = {
		limit = { check_variable = { global.num_days > TAI_ly_next_election } }
		TAI_ly_election = yes
		country_event = { id = TAI_gov.3 days = 1 }
	}
	if = {
		limit = { NOT = { check_variable = { TAI_gov_last_party = ruling_party } } }
		set_variable = { TAI_gov_last_party = ruling_party }
		if = { limit = { NOT = { check_variable = { TAI_gov_premier_party = ruling_party } } } country_event = { id = TAI_gov.1 days = 3 } }
	}
	if = {
		limit = { has_country_flag = TAI_ly_minority NOT = { has_country_flag = TAI_ly_noconf_cd } }
		random_list = {
			6 = { set_country_flag = { flag = TAI_ly_noconf_cd days = 365 } country_event = { id = TAI_gov.4 days = 2 } }
			94 = { }
		}
	}
	# v3.19 动态席次：党员跳槽（每月向支持度靠拢 3%）与零星倒戈
	TAI_ly_drift = yes
	random_list = {
		76 = { }
		12 = { set_temp_variable = { ly_shift = random } multiply_temp_variable = { ly_shift = 4 } add_to_temp_variable = { ly_shift = 1 } TAI_ly_shift_from_gov = yes }
		12 = { set_temp_variable = { ly_shift = random } multiply_temp_variable = { ly_shift = 4 } add_to_temp_variable = { ly_shift = 1 } TAI_ly_shift_to_gov = yes }
	}

}
