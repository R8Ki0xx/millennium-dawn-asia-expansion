# Scope: the revived CHI (TAI still exists)
ROC_setup_new_chi = {
	### 清理旧中共内容
	ROC_clear_ccp_dynamic_modifiers = yes
	set_country_flag = CHI_local_debt_resolved
	clr_country_flag = CHI_ic_initialized
	ROC_clear_ccp_missions = yes
	# v3.10 bop / 港澳决议
	if = { limit = { has_power_balance = { id = CHI_factional_balance } } remove_power_balance = { id = CHI_factional_balance } }
	set_power_balance = { id = TAI_roc_faction_balance left_side = TAI_roc_old_guard right_side = TAI_roc_newcomers set_value = -0.4 }
	set_country_flag = TAI_hk_macau_unlocked
	clear_variable = CHI_korea_peninsula_tension
	set_country_flag = ROC_new_chi
	set_cosmetic_tag = CHI_ROC

	### 精神与法律
	ROC_sync_ideas_from_TAI = yes
	TAI_end_ccp_nko_aid = yes

	### 三军精神与 debuff（v3.3）
	ROC_copy_military_from_TAI = yes

	### 科技
	inherit_technology = TAI

	### 经济变量
	set_variable = { treasury = TAI.treasury }
	set_variable = { debt = TAI.debt }
	set_variable = { int_investments = TAI.int_investments }
	set_variable = { corporate_tax_rate = TAI.corporate_tax_rate }
	set_variable = { population_tax_rate = TAI.population_tax_rate }
	set_variable = { gdp_per_capita = TAI.gdp_per_capita }
	set_variable = { gdpc_converging_var = TAI.gdpc_converging_var }
	set_variable = { currency_strength = TAI.currency_strength }
	set_variable = { inflation_rate_var = TAI.inflation_rate_var }
	set_variable = { cb_policy_rate = TAI.cb_policy_rate }
	set_variable = { EA_hidden_labor_var = TAI.EA_hidden_labor_var }
	set_variable = { EA_rural_labor_var = TAI.EA_rural_labor_var }
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = EA_labor_market_modifier } } }
		add_dynamic_modifier = { modifier = EA_labor_market_modifier }
	}

	### 影响力
	set_variable = { domestic_influence_amount = TAI.domestic_influence_amount }
	clear_array = influence_array
	clear_array = influence_array_val
	for_loop_effect = {
		end = TAI.influence_array^num
		value = v
		add_to_array = { influence_array = TAI.influence_array^v }
		add_to_array = { influence_array_val = TAI.influence_array_val^v }
	}
	recalculate_influence = yes

	### 内部派系
	set_temp_variable = { nation_to_copy_from = TAI.id }
	copy_internal_factions = yes
	set_temp_variable = { nation_to_copy_from = TAI.id }
	copy_subideolgy_leader_throughput = yes

	### 政治
	if = { limit = { TAI = { has_government = democratic } } set_politics = { ruling_party = democratic elections_allowed = yes } }
	else_if = { limit = { TAI = { has_government = neutrality } } set_politics = { ruling_party = neutrality elections_allowed = yes } }
	else_if = { limit = { TAI = { has_government = nationalist } } set_politics = { ruling_party = nationalist elections_allowed = no } }
	else_if = { limit = { TAI = { has_government = fascism } } set_politics = { ruling_party = fascism elections_allowed = no } }
	else = { set_politics = { ruling_party = communism elections_allowed = no } }
	for_loop_effect = {
		end = TAI.party_pop_array^num
		value = v
		set_variable = { party_pop_array^v = TAI.party_pop_array^v }
	}
	# v3.16 popularities：千禧每月按引擎的意识形态支持度缩放各党，所以引擎支持度也要照台湾复制，
	#   否则中共时期的共产主义支持度会全部落到农工党（槽位 5）头上（照千禧 replicate_configure_politics 的写法）
	set_temp_variable = { roc_pop_dem = TAI.party_popularity@democratic }
	multiply_temp_variable = { roc_pop_dem = 100 }
	set_temp_variable = { roc_pop_com = TAI.party_popularity@communism }
	multiply_temp_variable = { roc_pop_com = 100 }
	set_temp_variable = { roc_pop_neu = TAI.party_popularity@neutrality }
	multiply_temp_variable = { roc_pop_neu = 100 }
	set_temp_variable = { roc_pop_fas = TAI.party_popularity@fascism }
	multiply_temp_variable = { roc_pop_fas = 100 }
	set_temp_variable = { roc_pop_nat = TAI.party_popularity@nationalist }
	multiply_temp_variable = { roc_pop_nat = 100 }
	set_popularities = { democratic = roc_pop_dem communism = roc_pop_com neutrality = roc_pop_neu fascism = roc_pop_fas nationalist = roc_pop_nat }
	add_popularity = { ideology = communism popularity = 0.03 }
	# v3.11 kmt left：民革（国民党左派）与劳动党在大陆工农中的基本盘
	add_to_variable = { party_pop_array^3 = 0.12 }
	add_to_variable = { party_pop_array^5 = 0.03 }
	# v3.19 parties：政党合法化、共产党 35%
	TAI_roc_legalize_parties = yes
	TAI_roc_ccp_35 = yes
	add_up_party_totals = yes
	set_party_positions = yes
	# 领导人轮替计数沿用台湾时期（set_leader_ROC 据此选出同一位国民党/亲民党领导人）
	set_variable = { conservatism_leader = TAI.conservatism_leader }
	set_variable = { Conservative_leader = TAI.Conservative_leader }
	set_variable = { Neutral_conservatism_leader = TAI.Neutral_conservatism_leader }
	set_variable = { Nat_Autocracy_leader = TAI.Nat_Autocracy_leader }
	set_temp_variable = { rul_party_temp = TAI.ruling_party }
	set_temp_variable = { change_leader_temp = 1 }
	change_ruling_party_effect = yes

	### 政治点数 / 稳定 / 战争支持
	set_temp_variable = { roc_diff = TAI.political_power }
	subtract_from_temp_variable = { roc_diff = political_power }
	add_political_power = roc_diff
	set_temp_variable = { roc_diff = TAI.has_stability }
	subtract_from_temp_variable = { roc_diff = has_stability }
	add_stability = roc_diff
	set_temp_variable = { roc_diff = TAI.has_war_support }
	subtract_from_temp_variable = { roc_diff = has_war_support }
	add_war_support = roc_diff

	### 剧情旗标
	set_country_flag = TAI_mainland_restored
	if = { limit = { TAI = { has_country_flag = TAI_pan_blue_path } } set_country_flag = TAI_pan_blue_path }
	if = { limit = { TAI = { has_country_flag = TAI_us_backing_flag } } set_country_flag = TAI_us_backing_flag }
}
