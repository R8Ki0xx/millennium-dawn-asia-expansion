		}
	}

VEL_CHI_rally = {
	if = {
		limit = { check_variable = { VEL_CHI_sel > 0 } }
		set_temp_variable = { vel_r = VEL_CHI_sel }
		subtract_from_temp_variable = { vel_r = 1 }
		if = { limit = { check_variable = { VEL_CHI_camp^vel_r < 5 } } add_to_variable = { VEL_CHI_camp^vel_r = 1 } }
		VEL_CHI_refresh_sel = yes
	}
}

VEL_CHI_set_main_issue = {
	for_loop_effect = { start = 0 end = 6 value = vi set_variable = { VEL_CHI_sal^vi = 1 } }
	set_temp_variable = { vel_mi = vel_issue }
	subtract_from_temp_variable = { vel_mi = 1 }
	set_variable = { VEL_CHI_sal^vel_mi = 2 }
	set_variable = { VEL_CHI_main = vel_issue }
	VEL_CHI_run_poll = yes
}

VEL_CHI_reset = {

	for_loop_effect = { start = 0 end = 32 value = vr set_variable = { VEL_CHI_rev^vr = 0 } set_variable = { VEL_CHI_camp^vr = 0 } }
	for_loop_effect = { start = 0 end = 6 value = vi set_variable = { VEL_CHI_sal^vi = 1 } }
	for_loop_effect = { start = 0 end = 7 value = vp set_variable = { VEL_CHI_cand^vp = 0 } set_variable = { VEL_CHI_run_seat^vp = 0 } set_variable = { VEL_CHI_run_nat^vp = 0 } }
	set_variable = { VEL_CHI_run_w = 0 }
	set_variable = { VEL_CHI_main = 0 }
	set_variable = { VEL_CHI_poll_ready = 0 }
	set_variable = { VEL_CHI_done = 0 }
	set_variable = { VEL_CHI_anim_n = 0 }

}

# 每天由 on_action 调用；每 30 天重算一次民调（开票进行中不刷新）
VEL_CHI_poll_tick = {
	if = { limit = { has_country_flag = VEL_CHI_inited } add_to_variable = { VEL_CHI_news_t = 1 } if = { limit = { check_variable = { VEL_CHI_news_t > 3 } } set_variable = { VEL_CHI_news_t = 0 } add_to_variable = { VEL_CHI_news_i = 1 } if = { limit = { check_variable = { VEL_CHI_news_i > 3 } } set_variable = { VEL_CHI_news_i = 0 } } } }
	if = {
		limit = { has_country_flag = VEL_CHI_inited check_variable = { global.num_days > VEL_CHI_poll_due } NOT = { check_variable = { VEL_CHI_phase = 2 } } }
		set_variable = { VEL_CHI_poll_due = global.num_days }
		add_to_variable = { VEL_CHI_poll_due = 30 }
		VEL_CHI_run_poll = yes
	}
}
